+import { parseVersion, versionOrder } from './version.js';
+
+const COMPARATORS = new Map([
+  ['=', (c) => c === 0],
+  ['>=', (c) => c >= 0],
+  ['>', (c) => c > 0],
+  ['<=', (c) => c <= 0],
+  ['<', (c) => c < 0],
+]);
+const TERM = /^(>=|<=|>|<|=|\^|~)?(\S+)$/;
+
+const v = (major, minor, patch) => ({ major, minor, patch });
+
+/** The upper bound (exclusive) that a caret allows: the first non-zero number may not change. */
+function caretCeiling({ major, minor, patch }) {
+  if (major > 0) return v(major + 1, 0, 0);
+  if (minor > 0) return v(0, minor + 1, 0);
+  return v(0, 0, patch + 1);
+}
+
+/** A term as the [operator, version] bounds that must all hold. */
+function bounds(term) {
+  const m = TERM.exec(term);
+  if (!m) throw new RangeError(`not a range: ${term}`);
+  const base = parseVersion(m[2]);
+  switch (m[1]) {
+    case '^':
+      return [['>=', base], ['<', caretCeiling(base)]];
+    case '~':
+      return [['>=', base], ['<', v(base.major, base.minor + 1, 0)]];
+    default:
+      return [[m[1] ?? '=', base]];
+  }
+}
+
+/** True when the version, such as "1.4.2" or "v1.4.2", is inside the range; throws a RangeError for either unreadable. */
+export function satisfies(version, range) {
+  const target = parseVersion(version);
+  if (typeof range !== 'string') throw new RangeError(`not a range: ${range}`);
+  return range
+    .trim()
+    .split(/\s+/)
+    .flatMap(bounds)
+    .every(([op, bound]) => COMPARATORS.get(op)(versionOrder(target, bound)));
+}
